--- LEADS TABLE ---
id | int | Null:NO | Default:null
job_id | int | Null:NO | Default:null
tradesperson_id | int | Null:NO | Default:null
message | text | Null:NO | Default:null
price_estimate | decimal(10,2) | Null:YES | Default:null
